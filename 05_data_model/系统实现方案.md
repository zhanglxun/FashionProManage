## 🎯 项目概述

### 项目目标
实现一个高效、可扩展的企业关系管理系统，支持FashionProManage平台中企业间的复杂业务关系管理。

### 技术栈
- **数据库**: MySQL 8.0+
- **缓存**: Redis
- **后端框架**: Spring Boot / Node.js
- **前端框架**: React / Vue.js
- **监控**: Prometheus + Grafana

### 核心功能
- 企业基础信息管理
- 企业关系网络管理
- 数据权限控制
- 关系分析统计

```sql
-- 企业主表（已合并优化）
CREATE TABLE `o_enterprise` (
  -- 基础字段
  `id` bigint NOT NULL AUTO_INCREMENT,
  `name` varchar(64) NOT NULL,
  `type` int DEFAULT NULL COMMENT '1-平台客户 2-代理商 3-供应商 4-工厂 5-综合供应商',
  `status` int NOT NULL DEFAULT 1,

  -- 扩展字段（合并自供应商/工厂表）
  `supplier_type` int DEFAULT NULL COMMENT '供应商类型',
  `factory_type` varchar(64) DEFAULT NULL COMMENT '专长工艺',
  `production_capacity` varchar(64) DEFAULT NULL COMMENT '产能描述',
  `min_order_quantity` int DEFAULT NULL COMMENT '最小订单数量',
  `lead_time_days` int DEFAULT NULL COMMENT '标准交期',
  `quality_rating` decimal(3,2) DEFAULT NULL COMMENT '质量评分',
  `delivery_rating` decimal(3,2) DEFAULT NULL COMMENT '交期评分',
  `service_rating` decimal(3,2) DEFAULT NULL COMMENT '服务评分',
  `business_level` tinyint DEFAULT NULL COMMENT '业务等级',
  `cooperation_level` tinyint DEFAULT NULL COMMENT '合作等级',
  `payment_method` tinyint DEFAULT NULL COMMENT '付款方式',
  `payment_terms` varchar(200) DEFAULT NULL COMMENT '付款条件',
  `cooperation_start_date` datetime DEFAULT NULL COMMENT '合作开始日期',

  -- 标准字段
  `create_time` datetime DEFAULT CURRENT_TIMESTAMP,
  `modify_time` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_enterprise_type_status` (`type`, `status`),
  KEY `idx_enterprise_name` (`name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 企业关系表（核心表）
CREATE TABLE `o_enterprise_relation` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `from_enterprise_id` bigint NOT NULL COMMENT '关系发起企业ID',
  `to_enterprise_id` bigint NOT NULL COMMENT '关系接收企业ID',
  `relation_type` int NOT NULL COMMENT '关系类型：1-代理 2-供应 3-客户 4-合作 5-上下级 6-投资 7-竞争 8-其他',
  `business_type` int DEFAULT NULL COMMENT '业务类型',
  `status` tinyint NOT NULL DEFAULT 1 COMMENT '1-正常 2-暂停 3-终止 4-待确认',
  `description` varchar(512) DEFAULT NULL COMMENT '关系描述',
  `create_time` datetime DEFAULT CURRENT_TIMESTAMP,
  `modify_time` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_relation_bidirectional` (`from_enterprise_id`, `to_enterprise_id`, `status`),
  KEY `idx_relation_type_status` (`relation_type`, `status`),
  CONSTRAINT `fk_relation_from` FOREIGN KEY (`from_enterprise_id`) REFERENCES `o_enterprise` (`id`) ON DELETE CASCADE,
  CONSTRAINT `fk_relation_to` FOREIGN KEY (`to_enterprise_id`) REFERENCES `o_enterprise` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 代理商扩展表（用于平台权限控制）
CREATE TABLE `o_enterprise_agent` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `enterprise_id` bigint NOT NULL,
  `parent_id` bigint DEFAULT NULL COMMENT '父级ID，平台公司为0',
  `agent_type` int DEFAULT NULL COMMENT '代理商类型',
  `agent_area` varchar(200) DEFAULT NULL COMMENT '代理区域',
  `commission_rate` decimal(5,2) DEFAULT NULL COMMENT '佣金比例',
  `cooperation_start_date` datetime DEFAULT NULL,
  `cooperation_end_date` datetime DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `idx_agent_hierarchy` (`parent_id`, `enterprise_id`),
  FOREIGN KEY (`enterprise_id`) REFERENCES `o_enterprise` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```

#### 数据迁移脚本
```sql
-- ===== 数据迁移脚本 =====
-- 执行前请务必备份数据库

-- 1. 供应商表数据迁移（已合并到主表）
UPDATE o_enterprise e
JOIN o_enterprise_supplier s ON e.id = s.enterprise_id
SET
  e.supplier_type = s.supplier_type,
  e.production_capacity = s.production_capacity,
  e.min_order_quantity = s.min_order_quantity,
  e.lead_time_days = s.lead_time_days,
  e.quality_rating = s.quality_rating,
  e.delivery_rating = s.delivery_rating,
  e.service_rating = s.service_rating,
  e.business_level = s.level,
  e.cooperation_level = s.cooperation_level,
  e.payment_method = s.payment_method,
  e.payment_terms = s.payment_terms,
  e.cooperation_start_date = s.cooperation_start_date,
  e.type = CASE
    WHEN e.type = 4 AND s.supplier_type IS NOT NULL THEN 5  -- 工厂且有供应信息->综合供应商
    ELSE e.type
  END;

-- 2. 工厂表数据迁移（已合并到主表）
UPDATE o_enterprise e
JOIN o_enterprise_factory f ON e.id = f.enterprise_id
SET
  e.factory_type = f.fatory_type,
  e.production_capacity = COALESCE(f.production_capacity, e.production_capacity),
  e.min_order_quantity = COALESCE(f.min_order_quantity, e.min_order_quantity),
  e.lead_time_days = COALESCE(f.lead_time_days, e.lead_time_days),
  e.quality_rating = COALESCE(f.quality_rating, e.quality_rating),
  e.delivery_rating = COALESCE(f.delivery_rating, e.delivery_rating),
  e.service_rating = COALESCE(f.service_rating, e.service_rating),
  e.business_level = COALESCE(f.level, e.business_level),
  e.cooperation_level = COALESCE(f.cooperation_level, e.cooperation_level),
  e.payment_method = COALESCE(f.payment_method, e.payment_method),
  e.payment_terms = COALESCE(f.payment_terms, e.payment_terms),
  e.cooperation_start_date = COALESCE(f.cooperation_start_date, e.cooperation_start_date),
  e.type = CASE
    WHEN e.type = 3 AND f.fatory_type IS NOT NULL THEN 5  -- 供应商且有工厂信息->综合供应商
    ELSE e.type
  END;

-- 3. 确认综合供应商类型设置
UPDATE o_enterprise
SET type = 5  -- 综合供应商
WHERE (supplier_type IS NOT NULL OR factory_type IS NOT NULL)
  AND type IN (3, 4);  -- 原供应商或工厂类型

-- 4. 关系数据迁移
INSERT INTO o_enterprise_relation (
  from_enterprise_id, to_enterprise_id, relation_type, business_type,
  status, description, create_time, modify_time
)
SELECT
  r.agent_id,
  r.enterprise_id,
  CASE r.type
    WHEN 2 THEN 1  -- 代理商 → 代理关系
    WHEN 3 THEN 2  -- 供应商 → 供应关系
    WHEN 4 THEN 2  -- 工厂 → 供应关系
    ELSE 8        -- 其他 → 其他关系
  END as relation_type,
  r.business_type,
  1 as status,
  r.description,
  r.create_time,
  r.modify_time
FROM o_enterprise_relation_backup r
WHERE r.agent_id IS NOT NULL AND r.enterprise_id IS NOT NULL;

-- 5. 数据验证
SELECT '迁移后数据统计' as info;
SELECT
  COUNT(*) as total_enterprises,
  COUNT(CASE WHEN type = 3 THEN 1 END) as suppliers,
  COUNT(CASE WHEN type = 4 THEN 1 END) as factories,
  COUNT(CASE WHEN type = 5 THEN 1 END) as综合供应商
FROM o_enterprise;

SELECT '关系数据统计' as info;
SELECT
  COUNT(*) as total_relations,
  relation_type,
  COUNT(*) as count
FROM o_enterprise_relation
GROUP BY relation_type;
```

### 迁移验证脚本
```python
#!/usr/bin/env python3
# 数据迁移验证脚本

import mysql.connector
import logging
from datetime import datetime

class MigrationValidator:
    def __init__(self, db_config):
        self.db_config = db_config
        self.logger = self._setup_logger()

    def _setup_logger(self):
        logging.basicConfig(
            level=logging.INFO,
            format='%(asctime)s - %(levelname)s - %(message)s'
        )
        return logging.getLogger(__name__)

    def connect(self):
        return mysql.connector.connect(**self.db_config)

    def validate_data_integrity(self):
        """验证数据完整性"""
        try:
            conn = self.connect()
            cursor = conn.cursor()

            # 验证企业数据
            cursor.execute("SELECT COUNT(*) FROM o_enterprise")
            total_enterprises = cursor.fetchone()[0]
            self.logger.info(f"企业总数: {total_enterprises}")

            # 验证关系数据
            cursor.execute("SELECT COUNT(*) FROM o_enterprise_relation")
            total_relations = cursor.fetchone()[0]
            self.logger.info(f"关系总数: {total_relations}")

            # 验证外键约束
            cursor.execute("""
                SELECT COUNT(*) FROM o_enterprise_relation r
                LEFT JOIN o_enterprise e1 ON r.from_enterprise_id = e1.id
                LEFT JOIN o_enterprise e2 ON r.to_enterprise_id = e2.id
                WHERE e1.id IS NULL OR e2.id IS NULL
            """)
            invalid_relations = cursor.fetchone()[0]

            if invalid_relations > 0:
                self.logger.error(f"发现 {invalid_relations} 个无效的关系记录")
            else:
                self.logger.info("所有关系记录的外键约束验证通过")

            # 验证业务规则
            self._validate_business_rules(cursor)

            conn.close()
            return True

        except Exception as e:
            self.logger.error(f"数据验证失败: {e}")
            return False

    def _validate_business_rules(self, cursor):
        """验证业务规则"""

        # 检查自关联
        cursor.execute("""
            SELECT COUNT(*) FROM o_enterprise_relation
            WHERE from_enterprise_id = to_enterprise_id
        """)
        self_relations = cursor.fetchone()[0]

        if self_relations > 0:
            self.logger.warning(f"发现 {self_relations} 个自关联记录")

        # 检查重复关系
        cursor.execute("""
            SELECT from_enterprise_id, to_enterprise_id, relation_type, COUNT(*) as cnt
            FROM o_enterprise_relation
            WHERE status = 1
            GROUP BY from_enterprise_id, to_enterprise_id, relation_type
            HAVING cnt > 1
        """)
        duplicate_relations = cursor.fetchall()

        if duplicate_relations:
            self.logger.warning(f"发现 {len(duplicate_relations)} 组重复关系")

        # 检查孤立企业
        cursor.execute("""
            SELECT e.id, e.name
            FROM o_enterprise e
            LEFT JOIN o_enterprise_relation r1 ON e.id = r1.from_enterprise_id
            LEFT JOIN o_enterprise_relation r2 ON e.id = r2.to_enterprise_id
            WHERE r1.id IS NULL AND r2.id IS NULL
        """)
        orphan_enterprises = cursor.fetchall()

        if orphan_enterprises:
            self.logger.info(f"发现 {len(orphan_enterprises)} 个孤立企业")

    def performance_test(self):
        """性能测试"""
        try:
            conn = self.connect()
            cursor = conn.cursor()

            # 测试关系查询性能
            start_time = datetime.now()
            cursor.execute("""
                SELECT r.*, e1.name as from_name, e2.name as to_name
                FROM o_enterprise_relation r
                JOIN o_enterprise e1 ON r.from_enterprise_id = e1.id
                JOIN o_enterprise e2 ON r.to_enterprise_id = e2.id
                WHERE r.status = 1
                LIMIT 100
            """)
            query_time = (datetime.now() - start_time).total_seconds()
            self.logger.info(f"关系查询耗时: {query_time:.3f}秒")

            # 测试网络查询性能
            start_time = datetime.now()
            cursor.execute("""
                WITH RECURSIVE relation_network AS (
                    SELECT from_enterprise_id, to_enterprise_id, 1 as level
                    FROM o_enterprise_relation
                    WHERE from_enterprise_id = 1 AND status = 1
                    UNION ALL
                    SELECT r.from_enterprise_id, r.to_enterprise_id, rn.level + 1
                    FROM o_enterprise_relation r
                    JOIN relation_network rn ON r.from_enterprise_id = rn.to_enterprise_id
                    WHERE r.status = 1 AND rn.level < 3
                )
                SELECT COUNT(*) FROM relation_network
            """)
            network_query_time = (datetime.now() - start_time).total_seconds()
            self.logger.info(f"网络查询耗时: {network_query_time:.3f}秒")

            conn.close()

            return query_time < 1.0 and network_query_time < 2.0

        except Exception as e:
            self.logger.error(f"性能测试失败: {e}")
            return False

if __name__ == "__main__":
    db_config = {
        'host': 'localhost',
        'user': 'root',
        'password': 'password',
        'database': 'fashion_pro_manage'
    }

    validator = MigrationValidator(db_config)

    print("开始数据迁移验证...")

    if validator.validate_data_integrity():
        print("✅ 数据完整性验证通过")
    else:
        print("❌ 数据完整性验证失败")
        exit(1)

    if validator.performance_test():
        print("✅ 性能测试通过")
    else:
        print("❌ 性能测试失败")
        exit(1)

    print("🎉 所有验证测试通过，迁移成功！")
```

#### Q2: 如何处理循环关系？
**A**: 在创建关系时需要检查是否形成循环：
```java
public boolean wouldCreateCycle(Long fromId, Long toId) {
    // 使用递归查询检查路径
    Set<Long> visited = new HashSet<>();
    return hasPath(toId, fromId, visited);
}

private boolean hasPath(Long current, Long target, Set<Long> visited) {
    if (current == target) return true;
    if (visited.contains(current)) return false;

    visited.add(current);

    List<Relation> relations = relationRepository.findFromEnterprise(current);
    for (Relation relation : relations) {
        if (hasPath(relation.getToEnterpriseId(), target, visited)) {
            return true;
        }
    }

    return false;
}
```

#### Q3: 如何处理大数据量下的性能问题？

4. 查询优化：使用复合索引和分页查询

3. 乐观锁：更新操作使用版本号控制
4. 分布式锁：跨服务操作使用分布式锁
